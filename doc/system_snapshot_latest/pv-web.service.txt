# /etc/systemd/system/pv-web.service
[Unit]
Description=PV-System Web API (Gunicorn)
After=network-online.target pv-collector.service
Wants=network-online.target

[Service]
User=admin
WorkingDirectory=/home/primary-user/Dokumente/PVAnlage/pv-system
Environment=PATH=/home/primary-user/.local/bin:/usr/bin:/bin
ExecStartPre=-/bin/rm -f /tmp/pv_web.pid
ExecStart=/home/primary-user/.local/bin/gunicorn -c /home/primary-user/Dokumente/PVAnlage/pv-system/gunicorn_config.py web_api:app
Restart=always
RestartSec=10
StartLimitIntervalSec=600
StartLimitBurst=10
StandardOutput=journal
StandardError=journal

[Install]
WantedBy=multi-user.target

# /etc/systemd/system/pv-web.service.d/override.conf
[Service]
Environment=PV_MIRROR_MODE=0
