# /etc/systemd/system/pv-steuerbox.service
[Unit]
Description=PV-System Steuerbox (Operator-Intent API)
After=network-online.target pv-automation.service
Wants=network-online.target

[Service]
User=admin
WorkingDirectory=/home/primary-user/Dokumente/PVAnlage/pv-system
ExecStart=/usr/bin/python3 -m steuerbox.steuerbox_api
ExecReload=/bin/kill -HUP $MAINPID
Restart=always
RestartSec=20
StandardOutput=journal
StandardError=journal
Environment=PYTHONUNBUFFERED=1

[Install]
WantedBy=multi-user.target
